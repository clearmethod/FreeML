find_package(Threads REQUIRED)
add_custom_target(MatrixTests)

# Read the shared registry so new cases automatically become individual CTests.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS MatrixFunctions.cpp)
file(STRINGS MatrixFunctions.cpp matrix_cases REGEX "^MATRIX_TEST\\([A-Za-z0-9_]+\\)")

function(add_matrix_tests backend)
  set(target MatrixTests_${backend})
  add_executable(${target} MatrixFunctions.cpp)
  add_dependencies(MatrixTests ${target})
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}")
  target_compile_definitions(${target} PRIVATE MATRIX_TEST_${backend})
  target_link_libraries(${target} PRIVATE Threads::Threads)
  # Loss headers also expose the Metal specialization in CPU tests on macOS.
  if(APPLE)
    target_link_libraries(${target} PRIVATE "-framework Metal" "-framework Foundation")
  endif()
  # Keep library assertions enabled even when testing a Release build.
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()

  if(backend STREQUAL "Metal")
    set(shader_backend Metal)
  elseif(backend STREQUAL "DX11")
    target_compile_definitions(${target} PRIVATE NOMINMAX)
    target_link_libraries(${target} PRIVATE d3d11 d3dcompiler)
    set(shader_backend DirectX11)
  endif()

  if(DEFINED shader_backend)
    add_custom_target(${target}_shaders
      COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${PROJECT_SOURCE_DIR}/MatrixLibrary/GPU/${shader_backend}/Shaders"
        "${CMAKE_CURRENT_BINARY_DIR}/GPU/${shader_backend}/Shaders"
      VERBATIM)
    add_dependencies(${target} ${target}_shaders)
  endif()

  foreach(case IN LISTS matrix_cases)
    string(REGEX REPLACE "^MATRIX_TEST\\(([A-Za-z0-9_]+)\\).*" "\\1" case_name "${case}")
    add_test(NAME Matrix.${backend}.${case_name} COMMAND ${target} ${case_name})
    set_tests_properties(Matrix.${backend}.${case_name} PROPERTIES
      WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
      LABELS "matrix;${backend}" TIMEOUT 60 SKIP_RETURN_CODE 77)
  endforeach()
endfunction()

add_matrix_tests(CPU)
if(WIN32)
  add_matrix_tests(DX11)
endif()
if(APPLE)
  add_matrix_tests(Metal)
endif()

file(STRINGS LayerCases.def layer_cases REGEX "^LAYER_CASE\\([A-Za-z0-9_]+\\)")

function(add_layer_tests backend)
  set(target LayerTests_${backend})
  add_executable(${target} Layers.cpp)
  add_dependencies(LayerTests ${target})
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}")
  target_compile_definitions(${target} PRIVATE LAYER_TEST_${backend})
  target_link_libraries(${target} PRIVATE Threads::Threads)
  # Layer headers include the Metal bridge even for CPU instantiations on macOS.
  if(APPLE)
    target_link_libraries(${target} PRIVATE "-framework Metal" "-framework Foundation")
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
  if(backend STREQUAL "Metal")
    target_link_libraries(${target} PRIVATE "-framework Metal" "-framework Foundation")
    add_dependencies(${target} MatrixTests_Metal_shaders)
  elseif(backend STREQUAL "DX11")
    target_compile_definitions(${target} PRIVATE NOMINMAX)
    target_link_libraries(${target} PRIVATE d3d11 d3dcompiler)
    add_dependencies(${target} MatrixTests_DX11_shaders)
  endif()
  foreach(case IN LISTS layer_cases)
    string(REGEX REPLACE "^LAYER_CASE\\(([A-Za-z0-9_]+)\\).*" "\\1" case_name "${case}")
    foreach(pass Forward Backward)
      add_test(NAME Layer.${backend}.${case_name}.${pass} COMMAND ${target} ${case_name} ${pass})
      set_tests_properties(Layer.${backend}.${case_name}.${pass} PROPERTIES
        WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
        LABELS "layer;${backend};${pass}" TIMEOUT 180 SKIP_RETURN_CODE 77)
    endforeach()
  endforeach()
endfunction()
